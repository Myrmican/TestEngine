file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS "src/*.cpp" "include/*.h")

add_library(EngineCore OBJECT ${ENGINE_SOURCES})

target_compile_features(EngineCore PUBLIC cxx_std_20)
target_include_directories(EngineCore PUBLIC 
    "${CMAKE_CURRENT_SOURCE_DIR}/include/engine"
    "${CMAKE_SOURCE_DIR}/external"
    "${wasmtime_SOURCE_DIR}/include"
)

# reflection registry — same file, same directory scope
file(GLOB_RECURSE REFLECTION_HEADERS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/include/engine/datamodel/instances/*.h"
)
set(REFLECTION_REGISTRY_FILE "${CMAKE_CURRENT_BINARY_DIR}/generated/ReflectionRegistry.cpp")
set(_content "// Auto-generated by CMake — do not edit\n")
foreach(HEADER ${REFLECTION_HEADERS})
    string(APPEND _content "#include \"${HEADER}\"\n")
endforeach()
file(WRITE "${REFLECTION_REGISTRY_FILE}" "${_content}")

target_sources(EngineCore PRIVATE "${REFLECTION_REGISTRY_FILE}")   # <-- same directory as add_library
